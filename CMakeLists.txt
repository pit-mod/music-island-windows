cmake_minimum_required(VERSION 3.24)
project(MusicIslandWindows VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "MusicIsland requires Windows and MSVC with a Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(MusicIsland WIN32 src/Main.cpp src/Media.cpp src/Audio.cpp src/Renderer.cpp src/Keyboard.cpp app.rc)
target_compile_definitions(MusicIsland PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(MusicIsland PRIVATE /W4 /utf-8 /permissive- /Zc:__cplusplus)
target_link_options(MusicIsland PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
target_link_libraries(MusicIsland PRIVATE user32 gdi32 shell32 advapi32 ole32 oleaut32 uiautomationcore uuid windowsapp runtimeobject d3d11 dxgi d2d1 dwrite dcomp windowscodecs shlwapi mmdevapi)
add_executable(MotionTests tests/MotionTests.cpp)
target_compile_definitions(MotionTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
enable_testing()
add_test(NAME MotionTests COMMAND MotionTests)
add_executable(SpectrumTests tests/SpectrumTests.cpp)
target_compile_options(SpectrumTests PRIVATE /W4 /utf-8)
add_test(NAME SpectrumTests COMMAND SpectrumTests)
add_executable(SettingsTests tests/SettingsTests.cpp)
target_compile_definitions(SettingsTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(SettingsTests PRIVATE /W4 /utf-8)
target_link_libraries(SettingsTests PRIVATE shell32 ole32 uuid)
add_test(NAME SettingsTests COMMAND SettingsTests)
add_executable(AudioIntegrationTests tests/AudioIntegrationTests.cpp src/Audio.cpp)
target_compile_definitions(AudioIntegrationTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(AudioIntegrationTests PRIVATE /W4 /utf-8)
target_link_libraries(AudioIntegrationTests PRIVATE ole32 uuid mmdevapi)
# Run explicitly on Windows with an available playback endpoint; no audible test audio is emitted.

add_executable(KeyboardTests tests/KeyboardTests.cpp src/Keyboard.cpp)
target_compile_definitions(KeyboardTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(KeyboardTests PRIVATE /W4 /utf-8)
target_link_libraries(KeyboardTests PRIVATE user32 ole32 oleaut32 uiautomationcore uuid)
add_test(NAME KeyboardTests COMMAND KeyboardTests)
